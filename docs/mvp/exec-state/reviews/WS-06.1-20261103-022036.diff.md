+async def verify_event_tables(conn: Any, tables: Iterable[str] = EVENT_TABLES) -> None:
+    """Check that ``tables`` exist in the current schema with every column slot 010 creates.
+
+    Runs a single ``information_schema.columns`` query and never DDL. Raises
+    ``EventStoreSchemaError`` mapping each incomplete table to its missing columns (an absent
+    table lists all of them), and ``ValueError`` for a table that is not an event table.
+    """

+    def __init__(self, schema: dict[str, set[str]]) -> None:
+        self.schema = schema
+        self.sql: list[str] = []
+        self.args: list[tuple[Any, ...]] = []
+
+    async def __aenter__(self) -> _SchemaConn:
+        return self
+
+    async def __aexit__(self, *exc: Any) -> None:
+        return None
+
+    async def fetch(self, sql: str, *args: Any) -> list[dict[str, str]]:
+        self.sql.append(sql)
+        self.args.append(args)
+        if "information_schema.columns" not in sql:
+            return []
+        tables = args[0] if args else list(self.schema)
+        return [{"table_name": t, "column_name": c} for t in tables for c in sorted(self.schema.get(t, ()))]
+
+    async def execute(self, sql: str, *args: Any) -> str:
+        self.sql.append(sql)
+        self.args.append(args)
+        return "OK"
+
+    async def fetchrow(self, sql: str, *args: Any) -> None:
+        self.sql.append(sql)
+        self.args.append(args)
+        return None
