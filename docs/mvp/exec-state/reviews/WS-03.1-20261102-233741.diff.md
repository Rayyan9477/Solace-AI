+"""Hermetic ``docker compose config`` rendering for the env-contract tests and the harness.
+
+``render_compose_config`` copies the named compose files into ``workdir``, writes the given
+values to ``workdir/render.env`` (every value single-quoted, so compose never interpolates
+inside them), creates a ``workdir/.env.prod`` placeholder (older files that still say
+``env_file: .env.prod`` render without a real one ever being read), and runs::
+
+    docker compose --project-directory <workdir> -p solace-render --env-file <workdir>/render.env \\
+        -f <file> ... [--profile <p> ...] config --format json
+
+with a scrubbed process environment, so neither the developer's ``.env`` nor their shell
+variables can leak into the render. The rendered configuration is returned, never printed.
+
+Two normalizations of compose's output are undone so the result says what a container gets:
+
+- compose writes a literal ``$`` in a value as ``$$``; ``environment`` values are returned
+  unescaped (a bcrypt hash renders as itself). Other fields keep compose's escaping.
+- compose inlines ``env_file`` contents into ``environment`` and drops the ``env_file`` key, so
+  the placeholder holds a single probe variable: every service whose environment received it
+  is reported with ``env_file: [".env.prod"]`` and the probe is removed. Any other
+  ``env_file`` fails the render, because no such file exists in ``workdir``.
+"""
+from __future__ import annotations
+
+import json
+import os
+import re
+import shutil
+import subprocess
+from pathlib import Path
+from typing import Mapping, Sequence
+
+REPO_ROOT = Path(__file__).resolve().parents[2]
+PROJECT_NAME = "solace-render"
+RENDER_TIMEOUT_SECONDS = 180
+
+# The only parent-environment variables the docker CLI sees (compared case-insensitively,
+# because Windows environment names are case-insensitive). ProgramFiles is where Docker Desktop
+# installs the compose CLI plugin on Windows; without it ``docker compose`` is not found.
+PASSTHROUGH_ENV = (
+    "PATH", "SYSTEMROOT", "TEMP", "TMP", "USERPROFILE", "HOME", "APPDATA", "LOCALAPPDATA",
+    "ProgramData", "ProgramFiles", "DOCKER_HOST", "DOCKER_CONFIG", "DOCKER_CONTEXT",
+)
+
+ENV_FILE_PLACEHOLDER = ".env.prod"
+ENV_FILE_PROBE = "SOLACE_RENDER_ENV_FILE_PROBE"
+
+_KEY = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*$")
+_FORBIDDEN_IN_VALUE = ("'", "\n", "\r", "\x00")
+
+
+class ComposeRenderError(RuntimeError):
+    """``docker compose config`` failed; the message is compose's stderr (never a value)."""
+
+
+def format_env_file(values: Mapping[str, str]) -> str:
+    """Render ``values`` as a compose env file with one single-quoted ``KEY='value'`` per line.
+
+    Single quotes make compose take the value literally (no ``$`` interpolation). A value that
+    cannot be single-quoted (it holds a quote or a line break) is refused without echoing it.
+    """
+    lines = []
+    for key in sorted(values):
+        value = values[key]
+        if not _KEY.match(key):
+            raise ValueError(f"invalid environment variable name: {key!r}")
+        if not isinstance(value, str):
+            raise TypeError(f"{key}: value must be a string")
+        if any(ch in value for ch in _FORBIDDEN_IN_VALUE):
+            raise ValueError(f"{key}: value cannot be written single-quoted (quote or line break)")
+        lines.append(f"{key}='{value}'")
+    return "\n".join(lines) + "\n"
+
+
+def _scrubbed_env() -> dict[str, str]:
+    allowed = {name.upper() for name in PASSTHROUGH_ENV}
+    return {k: v for k, v in os.environ.items() if k.upper() in allowed}
+
+
+def _resolve(file: str | os.PathLike[str]) -> Path:
+    path = Path(file)
+    return path if path.is_absolute() else REPO_ROOT / path
+
+
+def render_compose_config(
+    files: Sequence[str | os.PathLike[str]],
+    env: Mapping[str, str],
+    *,
+    profiles: Sequence[str] = (),
+    workdir: Path,
+) -> dict:
+    """Render the merged compose configuration of ``files`` under ``env`` and return it as a dict.
+
+    ``files`` are repository-relative (or absolute) compose file paths, merged in order.
+    Raises ``ComposeRenderError`` when the docker CLI is missing or compose exits non-zero
+    (for example when a ``${VAR:?message}`` variable is unset).
+    """
+    if not files:
+        raise ValueError("at least one compose file is required")
+    docker = shutil.which("docker")
+    if docker is None:
+        raise ComposeRenderError("docker CLI not found on PATH")
+
+    workdir = Path(workdir)
+    workdir.mkdir(parents=True, exist_ok=True)
+    copied: list[Path] = []
+    for file in files:
+        source = _resolve(file)
+        target = workdir / source.name
+        if target in copied:
+            raise ValueError(f"two compose files share the name {source.name}")
+        shutil.copyfile(source, target)
+        copied.append(target)
+
+    env_file = workdir / "render.env"
+    env_file.write_text(format_env_file(env), encoding="utf-8")
+    (workdir / ENV_FILE_PLACEHOLDER).write_text(f"{ENV_FILE_PROBE}=1\n", encoding="utf-8")
+
+    cmd = [docker, "compose", "--project-directory", str(workdir), "-p", PROJECT_NAME,
+           "--env-file", str(env_file)]
+    for path in copied:
+        cmd += ["-f", str(path)]
+    for profile in profiles:
+        cmd += ["--profile", profile]
+    cmd += ["config", "--format", "json"]
+
+    try:
+        proc = subprocess.run(
+            cmd, cwd=workdir, env=_scrubbed_env(), capture_output=True, text=True,
+            encoding="utf-8", errors="replace", timeout=RENDER_TIMEOUT_SECONDS, check=False,
+        )
+    except subprocess.TimeoutExpired as exc:
+        raise ComposeRenderError(f"docker compose config timed out after {exc.timeout}s") from None
+    if proc.returncode != 0:
+        raise ComposeRenderError(proc.stderr.strip() or f"docker compose config exited {proc.returncode}")
+    try:
+        config = json.loads(proc.stdout)
+    except json.JSONDecodeError as exc:
+        raise ComposeRenderError(f"docker compose config produced no JSON ({exc.msg})") from None
+    for service in (config.get("services") or {}).values():
+        environment = service.get("environment") or {}
+        for name, value in environment.items():
+            if isinstance(value, str):
+                environment[name] = value.replace("$$", "$")
+        if ENV_FILE_PROBE in environment:
+            del environment[ENV_FILE_PROBE]
+            service.setdefault("env_file", []).append(ENV_FILE_PLACEHOLDER)
+    return config

+"""Static (AST) scan of the environment variable names that code reads.
+
+Collected forms:
+
+- ``os.getenv("X")`` / ``getenv("X")``, ``os.environ.get("X")`` / ``environ.get("X")``,
+  ``os.environ["X"]`` (load context) and ``"X" in os.environ``;
+- pydantic-settings classes: for every class whose body assigns
+  ``model_config = SettingsConfigDict(env_prefix="P", ...)`` (or a pydantic v1 style
+  ``class Config: env_prefix = "P"``), ``P + FIELD.upper()`` for each annotated field, plus any
+  literal ``alias`` / ``validation_alias`` / ``AliasChoices(...)`` names on the field.
+
+Any directory named ``tests`` is skipped. Only string literals are resolved; dynamic names are
+invisible to this scan.
+"""
+from __future__ import annotations
+
+import ast
+from pathlib import Path
+from typing import Iterator, Sequence
+
+SKIP_DIRS = frozenset({"tests", "__pycache__", ".venv", "venv", "node_modules", ".git"})
+
+
+def _is_os_attr(node: ast.AST, attr: str) -> bool:
+    """True for ``os.<attr>`` or a bare ``<attr>`` name (``from os import environ``)."""
+    if isinstance(node, ast.Attribute):
+        return node.attr == attr and isinstance(node.value, ast.Name) and node.value.id == "os"
+    return isinstance(node, ast.Name) and node.id == attr
+
+
+def _str_const(node: ast.AST | None) -> str | None:
+    if isinstance(node, ast.Constant) and isinstance(node.value, str):
+        return node.value
+    return None
+
+
+def _call_name(node: ast.AST) -> str | None:
+    if isinstance(node, ast.Call):
+        func = node.func
+        if isinstance(func, ast.Name):
+            return func.id
+        if isinstance(func, ast.Attribute):
+            return func.attr
+    return None
+
+
+def _settings_prefix(cls: ast.ClassDef) -> str | None:
+    """The env prefix of a pydantic-settings class, or None when the class is not one."""
+    for stmt in cls.body:
+        targets: list[ast.AST] = []
+        value: ast.AST | None = None
+        if isinstance(stmt, ast.Assign):
+            targets, value = stmt.targets, stmt.value
+        elif isinstance(stmt, ast.AnnAssign) and stmt.value is not None:
+            targets, value = [stmt.target], stmt.value
+        if any(isinstance(t, ast.Name) and t.id == "model_config" for t in targets):
+            if isinstance(value, ast.Call) and _call_name(value) == "SettingsConfigDict":
+                for kw in value.keywords:
+                    if kw.arg == "env_prefix":
+                        return _str_const(kw.value) or ""
+                return ""
+        if isinstance(stmt, ast.ClassDef) and stmt.name == "Config":
+            for inner in stmt.body:
+                if isinstance(inner, ast.Assign) and any(
+                    isinstance(t, ast.Name) and t.id == "env_prefix" for t in inner.targets
+                ):
+                    return _str_const(inner.value) or ""
+    return None
+
+
+def _field_aliases(value: ast.AST | None) -> Iterator[str]:
+    if not isinstance(value, ast.Call) or _call_name(value) != "Field":
+        return
+    for kw in value.keywords:
+        if kw.arg not in ("alias", "validation_alias"):
+            continue
+        literal = _str_const(kw.value)
+        if literal:
+            yield literal
+        elif isinstance(kw.value, ast.Call) and _call_name(kw.value) == "AliasChoices":
+            for arg in kw.value.args:
+                choice = _str_const(arg)
+                if choice:
+                    yield choice
+
+
+def _settings_fields(cls: ast.ClassDef, prefix: str) -> Iterator[tuple[str, int]]:
+    for stmt in cls.body:
+        if not (isinstance(stmt, ast.AnnAssign) and isinstance(stmt.target, ast.Name)):
+            continue
+        field = stmt.target.id
+        if field == "model_config" or field.startswith("_"):
+            continue
+        annotation = ast.unparse(stmt.annotation)
+        if annotation.startswith(("ClassVar", "typing.ClassVar")):
+            continue
+        yield prefix + field.upper(), stmt.lineno
+        for alias in _field_aliases(stmt.value):
+            yield alias, stmt.lineno
+
+
+def _names_read(tree: ast.AST) -> Iterator[tuple[str, int]]:
+    for node in ast.walk(tree):
+        if isinstance(node, ast.Call):
+            func = node.func
+            first = _str_const(node.args[0]) if node.args else None
+            if first is None:
+                continue
+            if _is_os_attr(func, "getenv"):
+                yield first, node.lineno
+            elif isinstance(func, ast.Attribute) and func.attr == "get" and _is_os_attr(func.value, "environ"):
+                yield first, node.lineno
+        elif isinstance(node, ast.Subscript) and isinstance(node.ctx, ast.Load):
+            if _is_os_attr(node.value, "environ"):
+                key = _str_const(node.slice)
+                if key:
+                    yield key, node.lineno
+        elif isinstance(node, ast.Compare):
+            key = _str_const(node.left)
+            if key and any(isinstance(op, (ast.In, ast.NotIn)) for op in node.ops):
+                if any(_is_os_attr(c, "environ") for c in node.comparators):
+                    yield key, node.lineno
+        elif isinstance(node, ast.ClassDef):
+            prefix = _settings_prefix(node)
+            if prefix is not None:
+                yield from _settings_fields(node, prefix)
+
+
+def _python_files(root: Path) -> Iterator[Path]:
+    for path in sorted(root.rglob("*.py")):
+        if any(part in SKIP_DIRS for part in path.relative_to(root).parts[:-1]):
+            continue
+        yield path
+
+
+def collect_env_readers(roots: Sequence[Path]) -> dict[str, list[str]]:
+    """Map each env name read under ``roots`` to the sorted ``path:line`` locations that read it."""
+    readers: dict[str, set[str]] = {}
+    for root in roots:
+        root = Path(root)
+        if not root.is_dir():
+            continue
+        for path in _python_files(root):
+            try:
+                tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
+            except (SyntaxError, UnicodeDecodeError, ValueError):
+                continue
+            for name, line in _names_read(tree):
+                readers.setdefault(name, set()).add(f"{path.as_posix()}:{line}")
+    return {name: sorted(locs) for name, locs in sorted(readers.items())}

+def _as_text(value: Any) -> str | None:
+    if value is None:
+        return None
+    if isinstance(value, bool):
+        return "true" if value else "false"
+    if isinstance(value, (int, float, str)):
+        return str(value)

+def _as_tuple(value: Any, what: str, where: str) -> tuple[str, ...]:
+    if isinstance(value, str) or not isinstance(value, (list, tuple)):
+        raise RegistryError(f"{where}: {what} must be a list")
+    if not all(isinstance(v, str) and v for v in value):

+def _row(raw: Any, index: int) -> EnvVar:

+    unknown = set(raw) - _FIELD_NAMES

+    where = f"{where} ({name})"

+            kwargs[key] = raw[key]

+    if not row.group:

+def _check_consistency(rows: list[EnvVar]) -> None:
+    seen: set[tuple[str, str]] = set()
+    by_source: dict[str, EnvVar] = {}
+    for row in rows:
+        for service in row.services:
+            if (row.name, service) in seen:
+                raise RegistryError(f"{row.name} is registered twice for {service}")
+            seen.add((row.name, service))
+        if row.source is None or row.retired:

+        if (first.secret, first.format) != (row.secret, row.format):

+def _pem_private_key() -> str:

+    from cryptography.hazmat.primitives.asymmetric import rsa

+    return pem.strip().replace("\n", "\\n")

+    """A fresh random value for a secret format, or ``example`` for any other format.

+    ``bcrypt`` yields a ``$2a$14$``-prefixed 60-character string with the bcrypt alphabet. It is
+    shaped like a hash but is not one; it exists for render tests only.

+        raise ValueError(f"unknown format {fmt!r}")
+    if example is None:
+        raise ValueError(f"format {fmt!r} has no generator; the registry row needs an example")
+    return example

+    groups: dict[str, list[tuple[str, EnvVar, list[EnvVar]]]] = {}
+    for source, rows in sources(list(registry)).items():
+        primary = _primary(source, rows)
+        groups.setdefault(primary.group, []).append((source, primary, rows))
+    lines = list(HEADER)
+    for group, entries in groups.items():
+        lines += ["", f"# --- {group} ---"]
+        for source, primary, rows in entries:
+            lines.append(_comment(primary, rows))
+            lines.append(f"{source}={_value(primary, rows)}")
+    return "\n".join(lines) + "\n"
+

+    parser = argparse.ArgumentParser(prog="python -m deploy.env.render_env_example", description=__doc__.split("\n")[0])
+    mode = parser.add_mutually_exclusive_group()
+    mode.add_argument("--write", action="store_true", help="rewrite the output file")
+    mode.add_argument("--check", action="store_true", help="exit 1 if the output file is stale")
+    parser.add_argument("--registry", type=Path, default=REGISTRY_PATH)
+    parser.add_argument("--output", type=Path, default=ENV_EXAMPLE_PATH)
+    args = parser.parse_args(argv)
+
+    text = render_env_example(load_registry(args.registry))
+    if args.write:
+        args.output.write_text(text, encoding="utf-8", newline="\n")
+        print(f"wrote {args.output}")
+        return 0
+    if args.check:
+        current = args.output.read_text(encoding="utf-8") if args.output.exists() else None
+        if current != text:
+            print(f"{args.output} is stale; run: python -m deploy.env.render_env_example --write", file=sys.stderr)
+            return 1
+        return 0
+    sys.stdout.write(text)
+    return 0
