+

+FastAPI app the same way:

+    async def evaluate_readiness(self) -> ReadinessReport:
+        """Run every readiness check concurrently (each with a timeout) and report.
+
+        A report is cached for ``READINESS_CACHE_SECONDS``. A failed check's ``detail`` is

+        ``solace_service_ready``.

+        if cached is not None:

+            checks = list(self._readiness_checks.items())

+            self._publish(report)

+            return None

+            return None

+    def _invalidate_readiness(self) -> None:
+        self._readiness_generation += 1
+        self._readiness_cache = None
+
+    @staticmethod
+    async def _run_check(entry: _RegisteredCheck) -> CheckResult:
+        try:
+            ok, detail = await asyncio.wait_for(

+            )
+        except Exception as exc:  # the class name only: the text may hold a DSN or secret
+            return CheckResult(ok=False, critical=entry.critical, detail=type(exc).__name__)
+        return CheckResult(ok=bool(ok), critical=entry.critical, detail=_safe_detail(detail))
+

+        service = self.spec.service_name

+            READINESS_CHECK_OK.labels(service=service, check=name).set(1 if result.ok else 0)
+        SERVICE_READY.labels(service=service).set(1 if report.ready else 0)

+"""Health routes for every service: ``/live``, ``/ready`` and ``/health`` (IF-01).
+
+- ``GET /live`` answers whether the process is healthy: 200 ``{"status": "alive"}``, or 503
+  ``{"status": "dead", "task": <name>}`` once a registered background task has exited.
+  It never touches a dependency, so a datastore outage never makes a container restart.
+- ``GET /ready`` (and ``GET /health``, the same answer) runs the registered readiness checks:
+  200 with ``status`` ``"ready"``, or 503 with ``"not_ready"`` when a critical check fails or
+  the runtime has not started yet.
+
+The routes read ``app.state.runtime`` (a ``ServiceRuntime``) and ``app.state.spec`` at request
+time; this module does not import the bootstrap, so it has no import cycle with it.
+"""
+
+from __future__ import annotations
+
+from collections.abc import Mapping
+from dataclasses import dataclass
+from typing import Any
+
+from fastapi import FastAPI, Request
+from fastapi.responses import JSONResponse
+
+from solace_infrastructure.runtime.metrics import SERVICE_READY
+
+LIVE_PATH = "/live"
+READY_PATH = "/ready"
+HEALTH_PATH = "/health"
+HEALTH_PATHS: frozenset[str] = frozenset({LIVE_PATH, READY_PATH, HEALTH_PATH})
+
+
+@dataclass(frozen=True)
+class CheckResult:
+    """One readiness check outcome. ``detail`` is fixed text or an exception class name."""
+
+    ok: bool
+    critical: bool
+    detail: str
+
+
+@dataclass(frozen=True)
+class ReadinessReport:
+    """The outcome of every readiness check. ``ready`` is False when a critical check failed."""
+
+    ready: bool
+    checks: Mapping[str, CheckResult]
+
+
+async def live_endpoint(request: Request) -> JSONResponse:
+    """Liveness: 503 naming the first registered background task that has exited."""
+    runtime = getattr(request.app.state, "runtime", None)
+    dead = runtime.dead_background_task() if runtime is not None else None
+    if dead is not None:
+        return JSONResponse({"status": "dead", "task": dead}, status_code=503)
+    return JSONResponse({"status": "alive"})
+
+
+async def ready_endpoint(request: Request) -> JSONResponse:
+    """Readiness: 200 when every critical check passes, else 503."""
+    spec = getattr(request.app.state, "spec", None)
+    service = getattr(spec, "service_name", "unknown")
+    body: dict[str, Any] = {
+        "status": "not_ready",
+        "service": service,
+        "version": getattr(spec, "version", "unknown"),
+        "checks": {},
+    }
+    runtime = getattr(request.app.state, "runtime", None)
+    if runtime is None:
+        # The lifespan has not started the runtime yet (or has torn it down).
+        SERVICE_READY.labels(service=service).set(0)
+        return JSONResponse(body, status_code=503)
+
+    report: ReadinessReport = await runtime.evaluate_readiness()
+    body["status"] = "ready" if report.ready else "not_ready"
+    body["checks"] = {
+        name: {"ok": result.ok, "critical": result.critical, "detail": result.detail}
+        for name, result in report.checks.items()
+    }
+    return JSONResponse(body, status_code=200 if report.ready else 503)
+
+
+def register_health_routes(app: FastAPI) -> None:
+    """Add ``GET /live``, ``GET /ready`` and ``GET /health`` to ``app`` (not in OpenAPI)."""
+    app.add_api_route(
+        LIVE_PATH, live_endpoint, methods=["GET"], include_in_schema=False, name="runtime_live"
+    )
+    app.add_api_route(
+        READY_PATH, ready_endpoint, methods=["GET"], include_in_schema=False, name="runtime_ready"
+    )
+    app.add_api_route(
+        HEALTH_PATH,
+        ready_endpoint,
+        methods=["GET"],
+        include_in_schema=False,
+        name="runtime_health",
+    )
+

+    "HEALTH_PATHS",
+    "LIVE_PATH",
+    "READY_PATH",
+    "CheckResult",

+    "register_health_routes",

+# Pydantic's own error messages are built from the schema, not from the input. These types,
+# and any custom error type, carry text a validator author wrote, which may quote the input.
+_AUTHOR_TEXT_ERROR_TYPES: frozenset[str] = frozenset({"value_error", "assertion_error"})
+_BUILTIN_ERROR_TYPES: frozenset[str] = frozenset(typing.get_args(ErrorType))
+_GENERIC_FIELD_MESSAGE = "Invalid value"
+_MAX_VALIDATION_DETAILS = 50
+_MAX_FIELD_LENGTH = 200
+# A location part that is not an identifier (for example a dict key the client sent, which
+# could be an e-mail address) is replaced, so the field path never echoes client data.
+_FIELD_PART_PATTERN = re.compile(r"[A-Za-z_][A-Za-z0-9_\-]{0,63}")
+_MASKED_FIELD_PART = "[key]"
+
+_log = structlog.get_logger("solace_infrastructure.runtime.http")

+# --------------------------------------------------------------------------- request ids
+
+
+def _load_phi_patterns() -> tuple[re.Pattern[str], ...]:
+    """The log sanitizer's PHI patterns, compiled the way its detector compiles them."""
+    try:
+        from solace_security.phi_protection import DEFAULT_PATTERNS
+    except ImportError:  # no sanitizer (development only): nothing to stay clear of
+        return ()
+    return tuple(re.compile(p.pattern, re.IGNORECASE) for p in DEFAULT_PATTERNS)
+
+
+# The ids are bound into every log line of the request, so they must pass the PHI log
+# sanitizer untouched: a masked id no longer correlates, and with the current sanitizer a
+# flagged context value makes every log call of the request recurse until RecursionError.
+_PHI_PATTERNS = _load_phi_patterns()
+
+
+def _looks_like_phi(value: str) -> bool:
+    return any(pattern.search(value) for pattern in _PHI_PATTERNS)
+
+
+def _mint_id() -> str:
+    """A fresh ``uuid4`` string the PHI log sanitizer does not flag (about 1 in 60 is)."""
+    candidate = str(uuid.uuid4())
+    for _ in range(_MAX_MINT_ATTEMPTS):
+        if not _looks_like_phi(candidate):
+            break
+        candidate = str(uuid.uuid4())
+    return candidate
+
+
+def _accept_or_mint(raw: bytes | None) -> str:
+    """Return the inbound id when it is well formed and not PHI-like, else mint one."""
+    if raw is not None:
+        try:
+            value = raw.decode("ascii")
+        except UnicodeDecodeError:
+            value = ""
+        if ID_PATTERN.fullmatch(value) and not _looks_like_phi(value):
+            return value
+    return _mint_id()
+
+
+def _inbound_ids(scope: Scope) -> dict[str, str]:
+    """Read the three id headers (first occurrence wins), minting any that are unusable."""
+    found: dict[bytes, bytes] = {}
+    for name, value in scope.get("headers") or ():
+        lowered = bytes(name).lower()
+        if lowered in _ID_HEADER_NAMES and lowered not in found:
+            found[lowered] = bytes(value)
+    return {key: _accept_or_mint(found.get(raw)) for key, _, raw in _ID_FIELDS}
+
+
+def _enter_request_context(ids: Mapping[str, str]) -> None:
+    structlog.contextvars.clear_contextvars()
+    clear_correlation_ids()
+    set_correlation_ids(
+        request_id=ids["request_id"],
+        correlation_id=ids["correlation_id"],
+        turn_id=ids["turn_id"],
+    )
+    structlog.contextvars.bind_contextvars(**ids)
+
+
+def _leave_request_context() -> None:
+    clear_correlation_ids()
+    structlog.contextvars.clear_contextvars()
+
+
+def _state_ids(request: Request) -> dict[str, str | None]:
+    """The ids the middleware stored for this request (``None`` when it did not run)."""
+    return {key: getattr(request.state, key, None) for key, _, _ in _ID_FIELDS}
+
+
+def _id_response_headers(ids: Mapping[str, str | None]) -> dict[str, str]:
+    return {header: value for key, header, _ in _ID_FIELDS if (value := ids.get(key))}
+
+
+def route_template(scope: Scope) -> str:
+    """The matched route's path template, or ``__unmatched__``. Never the raw request path.
+
+    FastAPI routes put themselves on ``scope["route"]``. Plain Starlette routes (``/docs``,
+    ``/openapi.json``) only set ``scope["endpoint"]``, so their template is looked up among
+    the app's routes by endpoint.
+    """
+    path = getattr(scope.get("route"), "path", None)
+    if isinstance(path, str) and path:
+        return path
+    endpoint = scope.get("endpoint")
+    if endpoint is not None:
+        router = getattr(scope.get("app"), "router", None)
+        for candidate in getattr(router, "routes", ()):
+            if getattr(candidate, "endpoint", None) is endpoint:
+                path = getattr(candidate, "path", None)
+                if isinstance(path, str) and path:
+                    return path
+    return UNMATCHED_ROUTE
+
+
+def _metric_method(method: Any) -> str:
+    upper = str(method or "").upper()
+    return upper if upper in _METRIC_METHODS else _OTHER_METHOD
+
+
+def quiet_outbound_url_logging() -> None:
+    """Hold the httpx loggers at WARNING or above, so no full outbound URL reaches the logs."""
+    for name in OUTBOUND_URL_LOGGERS:
+        outbound = logging.getLogger(name)
+        if outbound.level < logging.WARNING:  # NOTSET (0) included; a stricter level is kept
+            outbound.setLevel(logging.WARNING)

+class RequestContextMiddleware:
+    """Pure-ASGI request context: ids, response id headers, HTTP metrics and the access line."""
+
+    def __init__(self, app: ASGIApp, *, service_name: str) -> None:
+        self.app = app
+        self.service_name = service_name
+
+    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
+        if scope["type"] not in ("http", "websocket"):
+            await self.app(scope, receive, send)
+            return
+
+        ids = _inbound_ids(scope)
+        scope.setdefault("state", {}).update(ids)
+        _enter_request_context(ids)
+        try:
+            if scope["type"] == "http":
+                await self._serve_http(scope, receive, send, ids)
+            else:
+                await self.app(scope, receive, send)
+        finally:
+            _leave_request_context()
+
+    async def _serve_http(
+        self, scope: Scope, receive: Receive, send: Send, ids: Mapping[str, str]
+    ) -> None:
+        started = time.perf_counter()
+        status: int | None = None
+        id_headers = [
+            (raw, ids[key].encode("ascii")) for key, _, raw in _ID_FIELDS
+        ]
+
+        async def send_with_ids(message: Message) -> None:
+            nonlocal status
+            if message["type"] == "http.response.start":
+                status = int(message["status"])
+                kept = [
+                    (name, value)
+                    for name, value in message.get("headers") or ()
+                    if bytes(name).lower() not in _ID_HEADER_NAMES
+                ]
+                message = {**message, "headers": kept + id_headers}
+            await send(message)
+
+        try:
+            await self.app(scope, receive, send_with_ids)
+        except asyncio.CancelledError:
+            if status is None:
+                status = _CLIENT_CLOSED_STATUS
+            raise
+        except BaseException:
+            if status is None:
+                status = 500
+            raise
+        finally:
+            self._record(scope, 500 if status is None else status, time.perf_counter() - started)
+
+    def _record(self, scope: Scope, status: int, seconds: float) -> None:
+        route = route_template(scope)
+        method = _metric_method(scope.get("method"))
+        HTTP_REQUESTS.labels(
+            service=self.service_name, route=route, method=method, status=str(status)
+        ).inc()
+        HTTP_DURATION.labels(service=self.service_name, route=route).observe(seconds)
+        write = _log.debug if route in QUIET_ROUTES else _log.info
+        write(
+            "http_request",
+            service=self.service_name,
+            method=method,
+            route=route,
+            status=status,
+            duration_ms=round(seconds * 1000.0, 2),
+        )
+

+# --------------------------------------------------------------------------- error bodies
+
+
+def _error_body(
+    code: str, message: str, request_id: str | None, *, details: list[dict[str, str]] | None = None
+) -> dict[str, Any]:
+    error: dict[str, Any] = {"code": code, "message": message}
+    if details is not None:
+        error["details"] = details
+    error["request_id"] = request_id
+    return {"error": error}
+
+
+def _field_path(loc: Any) -> str:
+    parts: list[str] = []
+    for part in loc if isinstance(loc, list | tuple) else (loc,):
+        if isinstance(part, int):
+            parts.append(str(part))
+        elif isinstance(part, str) and _FIELD_PART_PATTERN.fullmatch(part):
+            parts.append(part)
+        else:
+            parts.append(_MASKED_FIELD_PART)
+    return ".".join(parts)[:_MAX_FIELD_LENGTH]
+
+
+def _validation_detail(error: Mapping[str, Any]) -> dict[str, str]:
+    """One error as ``{field, message, type}``; never the ``input`` or ``ctx`` values."""
+    error_type = str(error.get("type", "value_error"))
+    if error_type in _AUTHOR_TEXT_ERROR_TYPES or error_type not in _BUILTIN_ERROR_TYPES:
+        message = _GENERIC_FIELD_MESSAGE
+    else:
+        message = str(error.get("msg", _GENERIC_FIELD_MESSAGE))
+    return {"field": _field_path(error.get("loc", ())), "message": message, "type": error_type}
+
+
+async def _validation_error_handler(request: Request, exc: Exception) -> JSONResponse:
+    errors = exc.errors() if isinstance(exc, RequestValidationError) else []
+    details = [_validation_detail(error) for error in errors[:_MAX_VALIDATION_DETAILS]]
+    body = _error_body(
+        VALIDATION_ERROR_CODE,
+        VALIDATION_ERROR_MESSAGE,
+        getattr(request.state, "request_id", None),
+        details=details,
+    )
+    return JSONResponse(body, status_code=422)
+
+
+async def _websocket_validation_error_handler(websocket: WebSocket, exc: Exception) -> None:
+    # FastAPI's default puts the errors, input values included, into the close reason.
+    await websocket.close(code=WS_1008_POLICY_VIOLATION, reason=VALIDATION_ERROR_MESSAGE)
+
+
+async def _value_error_handler(request: Request, exc: Exception) -> JSONResponse:
+    if request.scope.get("type") != "http":
+        # A WebSocket has no response body to send; keep the default handling.
+        raise exc
+    _log.info("request_value_error", error_type=type(exc).__name__)
+    body = _error_body(BAD_REQUEST_CODE, BAD_REQUEST_MESSAGE, getattr(request.state, "request_id", None))
+    return JSONResponse(body, status_code=400)
+
+
+async def _unhandled_exception_handler(request: Request, exc: Exception) -> JSONResponse:
+    # Runs in ServerErrorMiddleware, outside RequestContextMiddleware: the contextvars are
+    # already cleared, so the ids come from request.state (they live on scope["state"]).
+    ids = _state_ids(request)
+    _log.error(
+        "unhandled_exception",
+        **ids,
+        method=_metric_method(request.scope.get("method")),
+        route=route_template(request.scope),
+        error_type=type(exc).__name__,
+        exc_info=exc,
+    )
+    body = _error_body(INTERNAL_ERROR_CODE, INTERNAL_ERROR_MESSAGE, ids["request_id"])
+    return JSONResponse(body, status_code=500, headers=_id_response_headers(ids))
+
+
+def install_exception_handlers(app: FastAPI, spec: ServiceRuntimeSpec) -> None:
+    """Install the uniform error bodies. No body ever carries ``str(exc)``.
+
+    - ``RequestValidationError`` -> 422 ``VALIDATION_ERROR`` with ``details`` of
+      ``{field, message, type}`` (no input values);
+    - ``ValueError`` -> 400 ``BAD_REQUEST``;
+    - any other exception -> 500 ``INTERNAL_ERROR`` (logged with its traceback);
+    - ``WebSocketRequestValidationError`` -> close 1008 with a fixed reason.
+
+    ``HTTPException`` keeps FastAPI's ``{"detail"}`` body; WS-04.9 (W2) adds the envelope for
+    ``spec.edge`` services.
+    """
+    del spec  # used from W2 (WS-04.9): edge services get the HTTPException envelope
+    app.add_exception_handler(RequestValidationError, _validation_error_handler)
+    app.add_exception_handler(WebSocketRequestValidationError, _websocket_validation_error_handler)
+    app.add_exception_handler(ValueError, _value_error_handler)
+    app.add_exception_handler(Exception, _unhandled_exception_handler)
+
+

+def cors_allowed_origins(spec: ServiceRuntimeSpec) -> list[str]:
+    """Origins from ``CORS_ALLOWED_ORIGINS`` (comma-separated).

+    Raises:
+        StartupDependencyError: in a strict environment, when the list is empty or holds
+            ``*`` or ``null``.

+                "cors", f"{CORS_ORIGINS_ENV} must not contain '*' or 'null' in strict environments"

+    return origins
+

+def install_cors(app: FastAPI, origins: list[str]) -> None:

+        expose_headers=list(CORS_EXPOSE_HEADERS),
+    )

+def _metric(cls: type, name: str, documentation: str, labelnames: tuple[str, ...]) -> Any:
+    """Create a metric on ``REGISTRY``, or return the one an earlier import registered.
+
+    A second import of this module (``importlib.reload`` in a test, or the same file reached
+    through a second package path) would otherwise fail with "Duplicated timeseries".
+    """
+    try:
+        return cls(name, documentation, labelnames)
+    except ValueError:
+        names_to_collectors = getattr(REGISTRY, "_names_to_collectors", {})
+        existing = names_to_collectors.get(name)
+        if isinstance(existing, cls) and tuple(getattr(existing, "_labelnames", ())) == labelnames:
+            return existing
+        raise
