+    "StartupDependencyError": _SPEC,

+    "is_strict_environment": "solace_security.production_guards",

+    from solace_infrastructure.runtime.spec import ServiceRuntimeSpec, StartupDependencyError
+    from solace_security.production_guards import is_strict_environment

+
+def __getattr__(name: str) -> Any:
+    module_name = _EXPORTS.get(name)
+    if module_name is None:
+        raise AttributeError(f"module {__name__!r} has no attribute {name!r}")
+    try:
+        module = importlib.import_module(module_name)
+    except ModuleNotFoundError as exc:
+        # The module that defines the name does not exist yet (bootstrap.py lands in a later
+        # task). Report a missing attribute so "from ... import name" fails as an ImportError
+        # naming the attribute. A missing dependency inside that module propagates unchanged.
+        if exc.name != module_name:
+            raise
+        raise AttributeError(
+            f"module {__name__!r} has no attribute {name!r} ({module_name} is not available)"
+        ) from exc
+    value = getattr(module, name)
+    globals()[name] = value
+    return value
+
+

+"""PHI-safe structured logging for every service (IF-01, MVP-180).
+
+``configure_service_logging(spec)`` installs one processor chain for structlog and a matching
+one for stdlib ``logging`` records (uvicorn, aiokafka, asyncpg and the rest). Both render one
+JSON object per line on stdout:
+
+    merge_contextvars, add_log_level, TimeStamper, StackInfoRenderer, format_exc_info,
+    phi_sanitizer_processor, JSONRenderer
+
+``format_exc_info`` turns ``exc_info`` into the ``exception`` traceback text before the PHI
+sanitizer runs. So ``logger.exception(...)`` logs the traceback, and the sanitizer sees that
+text (MVP-180). In strict environments the sanitizer is mandatory: if it cannot be imported,
+``StartupDependencyError("logging", ...)`` is raised and nothing is reconfigured. Outside
+strict environments it is left out and one warning says so.
+"""
+
+from __future__ import annotations
+
+import logging
+import sys
+from typing import Any
+
+import structlog
+from structlog.typing import Processor
+
+from solace_infrastructure.runtime.spec import ServiceRuntimeSpec, StartupDependencyError
+from solace_security.production_guards import is_strict_environment
+
+# Library loggers whose records go through the runtime handler on the root logger.
+ROUTED_STDLIB_LOGGERS: tuple[str, ...] = ("uvicorn", "uvicorn.error", "aiokafka", "asyncpg")
+
+# uvicorn.error logs WebSocket handshakes ('"WebSocket /path?token=..." [accepted]' and the
+# 403 lines) at INFO, query string included. It is kept at WARNING or above, so routing it
+# through this chain never brings those lines back after ``--log-level warning`` (R-01.2).
+UVICORN_ERROR_MIN_LEVEL: int = logging.WARNING
+
+# The runtime writes its own access line per request. uvicorn's access log carries raw query
+# strings, so it is disabled.
+SILENCED_STDLIB_LOGGERS: tuple[str, ...] = ("uvicorn.access",)
+
+_HANDLER_NAME = "solace-runtime-stdout"
+_DEFAULT_LEVEL = logging.INFO
+
+
+class _CurrentStdout:
+    """Stream for the root handler that writes to whatever ``sys.stdout`` is at write time.
+
+    The output is the same as ``StreamHandler(sys.stdout)``. Looking up ``sys.stdout`` on
+    every write means a replaced stdout (test capture, a re-opened stream) never leaves the
+    root logger writing to a stale or closed file.
+    """
+
+    def write(self, text: str) -> int:
+        return sys.stdout.write(text)
+
+    def flush(self) -> None:
+        sys.stdout.flush()
+
+
+def configure_service_logging(spec: ServiceRuntimeSpec) -> None:
+    """Install the runtime logging chain for structlog and for the stdlib root logger.
+
+    Safe to call more than once. Each call replaces the structlog configuration and the root
+    handler that an earlier call installed.
+
+    Raises:
+        StartupDependencyError: in a strict environment, when the PHI log sanitizer cannot be
+            imported.
+    """
+    strict = _is_strict(spec)
+    sanitizer, sanitizer_error = _load_phi_sanitizer(strict)
+    level, level_known = _resolve_level(spec.log_level)
+    sanitizers: list[Processor] = [sanitizer] if sanitizer is not None else []
+
+    renderer: Processor
+    if spec.debug and not strict:
+        # The traceback is already text (format_exc_info), so print it as is.
+        renderer = structlog.dev.ConsoleRenderer(exception_formatter=structlog.dev.plain_traceback)
+    else:
+        renderer = structlog.processors.JSONRenderer()
+
+    structlog.configure(
+        processors=[
+            structlog.contextvars.merge_contextvars,
+            structlog.processors.add_log_level,
+            structlog.processors.TimeStamper(fmt="iso", utc=True),
+            structlog.processors.StackInfoRenderer(),
+            structlog.processors.format_exc_info,
+            *sanitizers,
+            renderer,
+        ],
+        wrapper_class=structlog.make_filtering_bound_logger(level),
+        context_class=dict,
+        logger_factory=structlog.PrintLoggerFactory(),
+        # Not cached, so a later configure call reaches every logger and none keeps an older
+        # chain or a replaced stdout.
+        cache_logger_on_first_use=False,
+    )
+    _route_stdlib_logging(level, sanitizers)
+
+    log = structlog.get_logger(__name__)
+    if sanitizer_error is not None:
+        log.warning(
+            "phi_log_sanitizer_unavailable",
+            service=spec.service_name,
+            environment=spec.environment,
+            error_type=sanitizer_error,
+        )
+    if not level_known:
+        log.warning(
+            "log_level_unrecognised",
+            service=spec.service_name,
+            log_level=spec.log_level,
+            using=logging.getLevelName(_DEFAULT_LEVEL),
+        )
+
+
+def logging_chain_names() -> list[str]:
+    """Return the names of the processors in the active structlog configuration, in order."""
+    return [_processor_name(p) for p in structlog.get_config()["processors"]]
+
+
+def _is_strict(spec: ServiceRuntimeSpec) -> bool:
+    """Strict when the service's own environment or the deployment's ENVIRONMENT is strict."""
+    return is_strict_environment(spec.environment) or is_strict_environment()
+
+
+def _load_phi_sanitizer(strict: bool) -> tuple[Processor | None, str | None]:
+    """Return ``(sanitizer, None)``, or ``(None, error type)`` when it is unavailable outside strict."""
+    try:
+        from solace_security.phi_protection import phi_sanitizer_processor
+    except ImportError as exc:
+        if strict:
+            raise StartupDependencyError("logging", "PHI log sanitizer unavailable") from exc
+        return None, type(exc).__name__
+    return phi_sanitizer_processor, None
+
+
+def _resolve_level(name: str) -> tuple[int, bool]:
+    """Map a level name to its number; an unknown name gives INFO and ``False``."""
+    level = logging.getLevelNamesMapping().get(name.strip().upper())
+    if level is None:
+        return _DEFAULT_LEVEL, False
+    return level, True
+
+
+def _route_stdlib_logging(level: int, sanitizers: list[Processor]) -> None:
+    """Give the root logger one JSON handler on stdout and route the library loggers to it."""
+    formatter = structlog.stdlib.ProcessorFormatter(
+        foreign_pre_chain=[
+            structlog.processors.add_log_level,
+            structlog.processors.TimeStamper(fmt="iso", utc=True),
+            structlog.processors.StackInfoRenderer(),
+            structlog.processors.format_exc_info,
+        ],
+        processors=[
+            structlog.stdlib.ProcessorFormatter.remove_processors_meta,
+            *sanitizers,
+            structlog.processors.JSONRenderer(),
+        ],
+    )
+    handler = logging.StreamHandler(_CurrentStdout())
+    handler.set_name(_HANDLER_NAME)
+    handler.setFormatter(formatter)
+
+    root = logging.getLogger()
+    for existing in list(root.handlers):
+        if existing.get_name() == _HANDLER_NAME:
+            root.removeHandler(existing)
+            existing.close()
+    root.addHandler(handler)
+    root.setLevel(level)
+
+    for name in ROUTED_STDLIB_LOGGERS:
+        routed = logging.getLogger(name)
+        _remove_handlers(routed)
+        routed.setLevel(logging.NOTSET)
+        routed.propagate = True
+        routed.disabled = False
+    logging.getLogger("uvicorn.error").setLevel(max(level, UVICORN_ERROR_MIN_LEVEL))
+
+    for name in SILENCED_STDLIB_LOGGERS:
+        silenced = logging.getLogger(name)
+        _remove_handlers(silenced)
+        silenced.propagate = False
+        silenced.disabled = True
+
+
+def _remove_handlers(logger: logging.Logger) -> None:
+    for existing in list(logger.handlers):
+        logger.removeHandler(existing)
+
+
+def _processor_name(processor: Any) -> str:
+    # format_exc_info is a module-level ExceptionRenderer instance, so it has no __name__.
+    if processor is structlog.processors.format_exc_info:
+        return "format_exc_info"
+    name = getattr(processor, "__name__", None)
+    return name if isinstance(name, str) else type(processor).__name__
+
+
+__all__ = ["configure_service_logging", "logging_chain_names"]

+"""The service runtime declaration (IF-01) and the startup failure type.
+
+Each MVP service declares one frozen ``ServiceRuntimeSpec``. The runtime bootstrap reads the
+spec's flags to decide what to start (Postgres, Redis, the event plane, audit, rate limiting)
+and how strictly to treat a failed dependency. This module imports only the standard library,
+so declaring a spec never pulls in FastAPI, asyncpg or any client.
+"""
+
+from __future__ import annotations
+
+import re
+from dataclasses import dataclass
+from typing import Literal
+
+EVENT_PLANE_MODES: tuple[str, ...] = ("none", "publish", "publish_consume")
+
+# Any "scheme://..." run. A reason must never carry a DSN or URL, because a DSN can embed a
+# password; this pattern is the last line of defence when a caller gets that wrong.
+_URL_PATTERN = re.compile(r"[A-Za-z][A-Za-z0-9+.\-]*://\S*")
+_URL_PLACEHOLDER = "<redacted-url>"
+
+
+@dataclass(frozen=True)
+class ServiceRuntimeSpec:
+    """What one service needs from the shared runtime (IF-01).
+
+    ``environment`` is the service's own resolved setting (for example
+    ``settings.environment``). ``docs_enabled=None`` means ``/docs`` and ``/openapi.json``
+    are served only outside strict environments.
+    """
+
+    service_name: str
+    environment: str
+    version: str = "1.0.0"
+    log_level: str = "INFO"
+    debug: bool = False
+    uses_postgres: bool = False
+    uses_redis: bool = False
+    event_plane: Literal["none", "publish", "publish_consume"] = "none"
+    handles_phi: bool = False
+    audited: bool = False
+    rate_limited: bool = False
+    edge: bool = False
+    docs_enabled: bool | None = None
+
+    def __post_init__(self) -> None:
+        if self.event_plane not in EVENT_PLANE_MODES:
+            raise ValueError(
+                f"event_plane must be one of {EVENT_PLANE_MODES}, got {self.event_plane!r}"
+            )
+
+
+class StartupDependencyError(RuntimeError):
+    """A dependency the service needs at startup is missing or unusable.
+
+    ``component`` names the dependency (for example ``"logging"`` or ``"postgres"``) and
+    ``reason`` says what is wrong in fixed words. Callers never put a secret, DSN or URL into
+    ``reason``; any ``scheme://...`` text that slips through is replaced with
+    ``<redacted-url>`` before it is stored.
+    """
+
+    def __init__(self, component: str, reason: str) -> None:
+        self.component = component
+        self.reason = _URL_PATTERN.sub(_URL_PLACEHOLDER, reason)
+        super().__init__(self.component, self.reason)
+
+    def __str__(self) -> str:
+        return f"{self.component}: {self.reason}"
+
+
+__all__ = ["EVENT_PLANE_MODES", "ServiceRuntimeSpec", "StartupDependencyError"]

+"""MVP-180: logger.exception() must emit the traceback, and the traceback text must pass the PHI sanitizer."""
+import dataclasses, json, logging, subprocess, sys
+import pytest, structlog
+from solace_infrastructure.runtime import ServiceRuntimeSpec, StartupDependencyError
+from solace_infrastructure.runtime.logging import configure_service_logging, logging_chain_names
+
+def _spec(env="development", **kw):
+    return ServiceRuntimeSpec(service_name="safety-service", environment=env, **kw)
+
+def _last_json(out: str) -> dict:
+    return json.loads(out.strip().splitlines()[-1])
+
+def test_exception_log_carries_the_traceback(capsys):
+    configure_service_logging(_spec())
+    try:
+        raise RuntimeError("boom-xyz")
+    except RuntimeError:
+        structlog.get_logger("t").exception("crisis_handler_failed")
+    line = _last_json(capsys.readouterr().out)
+    assert line["event"] == "crisis_handler_failed"
+    assert "Traceback (most recent call last)" in line["exception"]
+    assert "RuntimeError" in line["exception"]
+    assert "exc_info" not in line
+
+def test_traceback_text_is_phi_sanitized(capsys):
+    configure_service_logging(_spec("production"))
+    try:
+        raise ValueError("lookup failed for jane.doe@example.com 555-867-5309")
+    except ValueError:
+        structlog.get_logger("t").exception("lookup_failed")
+    out = capsys.readouterr().out
+    assert "jane.doe@example.com" not in out
+    assert "555-867-5309" not in out
+
+def test_chain_order_puts_exc_info_before_sanitizer_before_renderer():
+    configure_service_logging(_spec("production"))
+    names = logging_chain_names()
+    assert names.index("format_exc_info") < names.index("phi_sanitizer_processor") < names.index("JSONRenderer")
+    assert names[:4] == ["merge_contextvars", "add_log_level", "TimeStamper", "StackInfoRenderer"]
+
+def test_strict_requires_the_phi_sanitizer(monkeypatch):
+    monkeypatch.setitem(sys.modules, "solace_security.phi_protection", None)
+    with pytest.raises(StartupDependencyError) as exc:
+        configure_service_logging(_spec("staging"))
+    assert exc.value.component == "logging"
