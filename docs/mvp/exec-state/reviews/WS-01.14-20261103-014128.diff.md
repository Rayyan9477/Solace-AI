+"""D-07 (SOPS + age): secret scan of every checkpoint-visible file. Stdlib plus PyYAML.
+
+``checkpoint.ps1`` snapshots the working tree with ``git add -A``; ``visible_paths`` lists exactly that set
+(``git ls-files --cached --others --exclude-standard``). Pre-commit hooks never run (D-17: no commits), so this
+scan runs as a ``tests/build`` test in every regression, barrier and ``make ci-local``.
+
+Rules. A finding names the rule and the path, never any file content:
+    age-private-key               the file's bytes hold age's private-key prefix followed by 50+ bech32 characters
+    age-key-file                  a file named ``*.agekey`` or ``keys.txt`` (any letter case)
+    stray-file-in-deploy-secrets  a file under ``deploy/secrets/`` that is not one of the planned files below
+    sops-metadata-missing         a ``*.sops.yaml`` other than the creation-rules file named exactly ``.sops.yaml``
+                                  that is not a YAML mapping, or has no top-level ``sops`` mapping holding a
+                                  non-empty ``age`` list and a ``mac``
+    sops-plaintext-value          in such a file, a scalar leaf outside the top-level ``sops`` mapping that is not a
+                                  string starting with ``ENC[AES256_GCM,``. The composed YAML node graph is walked,
+                                  so a duplicate key cannot hide a plaintext value.
+
+SOPS conventions (R-01.18; WS-03.8 follows them):
+    deploy/secrets/ holds only
+        <env>.sops.yaml                                  one per environment, encrypted to its age recipient(s)
+        .sops.yaml                                       SOPS creation rules, public recipients only
+        escrow-recipients.txt                            public recipients only
+        README.md                                        optional
+        history/<env>/<YYYYMMDDTHHMMSSZ>.sops.yaml       previous encrypted versions, named by UTC timestamp
+    <env> is lower-case: a letter, then letters, digits, ``-`` or ``_``.
+    Every <env>.sops.yaml is fully encrypted: no ``unencrypted_suffix``, ``unencrypted_regex`` or
+    ``encrypted_regex`` rule, and no key ending in SOPS's default ``_unencrypted`` suffix.
+    No ``*.agekey`` or ``keys.txt`` file is planned anywhere in the tree. Age private keys stay off-host and arrive at
+    deploy time through ``SOPS_AGE_KEY_FILE`` at a root-only path; decrypted values exist only on tmpfs /run/secrets.
+
+CLI:
+    secret_scan.py [--root DIR]   one finding per line on stdout; exit 1 on any finding, 0 when clean,
+                                  2 when the scan cannot run (git failure or an unreadable file)
+"""
+from __future__ import annotations
+
+import argparse
+import re
+import subprocess
+import sys
+from collections.abc import Iterable, Iterator
+from dataclasses import dataclass
+from pathlib import Path, PurePosixPath
+
+import yaml
+
+ROOT = Path(__file__).resolve().parents[2]
+
+# The prefix is never written contiguously, so this file, its tests and the plans never flag themselves.
+_AGE_PRIVATE_KEY = re.compile(re.escape(("AGE-SECRET-" + "KEY-1").encode()) + rb"[02-9AC-HJ-NP-Z]{50,}")
+_ENC_PREFIX = "ENC[AES256_GCM,"
+_STR_TAG = "tag:yaml.org,2002:str"
+_SECRETS_DIR = "deploy/secrets/"
+_CREATION_RULES = ".sops.yaml"
+_ENV = r"[a-z][a-z0-9_-]*"
+_PLANNED_SECRETS_FILE = re.compile(
+    rf"{_ENV}\.sops\.yaml|\.sops\.yaml|escrow-recipients\.txt|README\.md"
+    rf"|history/{_ENV}/[0-9]{{8}}T[0-9]{{6}}Z\.sops\.yaml"
+)
+
+
+@dataclass(frozen=True)
+class Finding:
+    """One rule hit on one checkpoint-visible file; carries names only, never content."""
+
+    rule: str
+    path: str
+
+    def __str__(self) -> str:
+        return f"{self.rule} {self.path}"
+
+
+def visible_paths(root: Path) -> list[str]:
+    """Every path ``git add -A`` would capture: tracked plus untracked, minus ignored (read-only git)."""
+    out = subprocess.run(
+        ["git", "-C", str(root), "ls-files", "--cached", "--others", "--exclude-standard", "-z"],
+        capture_output=True, check=True,
+    ).stdout
+    return sorted({p for p in out.decode("utf-8", "surrogateescape").split("\0") if p})
+
+
+def _has_sops_metadata(doc: dict) -> bool:
+    meta = doc.get("sops")
+    if not isinstance(meta, dict):
+        return False
+    age, mac = meta.get("age"), meta.get("mac")
+    return isinstance(age, list) and bool(age) and isinstance(mac, str) and bool(mac)
+
+
+def _has_plaintext_leaf(root: yaml.MappingNode) -> bool:
+    stack: list[yaml.Node] = [
+        value for key, value in root.value
+        if not (isinstance(key, yaml.ScalarNode) and key.tag == _STR_TAG and key.value == "sops")
+    ]
+    seen: set[int] = set()  # an alias points at an already-visited node; also stops recursive anchors
+    while stack:
+        node = stack.pop()
+        if id(node) in seen:
+            continue
+        seen.add(id(node))
+        if isinstance(node, yaml.MappingNode):
+            stack.extend(value for _, value in node.value)
+        elif isinstance(node, yaml.SequenceNode):
+            stack.extend(node.value)
+        elif not (node.tag == _STR_TAG and node.value.startswith(_ENC_PREFIX)):
+            return True
+    return False
+
+
+def _sops_rules(data: bytes) -> Iterator[str]:
+    try:
+        loader = yaml.SafeLoader(data)  # the reader decodes the first bytes here, so this can raise too
+        try:
+            node = loader.get_single_node()
+            doc = loader.construct_document(node) if node is not None else None
+        finally:
+            loader.dispose()
+    except (yaml.YAMLError, ValueError, RecursionError):
+        node = doc = None
+    if not isinstance(node, yaml.MappingNode) or not isinstance(doc, dict):
+        yield "sops-metadata-missing"
+        return
+    if not _has_sops_metadata(doc):
+        yield "sops-metadata-missing"
+    if _has_plaintext_leaf(node):
+        yield "sops-plaintext-value"
+
+
+def _rules(rel: str, data: bytes) -> Iterator[str]:
+    name = PurePosixPath(rel).name
+    lowered = name.lower()
+    if _AGE_PRIVATE_KEY.search(data):
+        yield "age-private-key"
+    if lowered.endswith(".agekey") or lowered == "keys.txt":
+        yield "age-key-file"
+    if rel.lower().startswith(_SECRETS_DIR) and not _PLANNED_SECRETS_FILE.fullmatch(rel[len(_SECRETS_DIR):]):
+        yield "stray-file-in-deploy-secrets"
+    if lowered.endswith(".sops.yaml") and name != _CREATION_RULES:
+        yield from _sops_rules(data)
+
+
+def scan(root: Path, paths: Iterable[str]) -> list[Finding]:
+    """Apply every rule to each existing file (read as bytes, so binaries are searched too).
+
+    Paths are repository-relative with ``/`` separators, as git prints them; a listed path that no longer exists
+    (deleted in the working tree) or is not a regular file is skipped. An unreadable file raises ``OSError``.
+    """
+    root = Path(root)
+    findings: list[Finding] = []
+    for raw in paths:
+        rel = raw.replace("\\", "/")
+        file = root / rel
+        if not file.is_file():
+            continue
+        findings.extend(Finding(rule, rel) for rule in _rules(rel, file.read_bytes()))
+    return findings
+
+
+def main(argv: list[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(description=__doc__.split("\n", 1)[0])
+    parser.add_argument("--root", type=Path, default=ROOT, help="repository to scan (default: this checkout)")
+    args = parser.parse_args(argv)
+    try:
+        paths = visible_paths(args.root)
+        findings = scan(args.root, paths)
+    except (OSError, subprocess.CalledProcessError) as exc:
+        print(f"secret_scan: cannot scan {args.root}: {exc}", file=sys.stderr)
+        return 2
+    for finding in findings:
+        print(finding)
+    print(f"secret_scan: {len(paths)} checkpoint-visible paths, {len(findings)} findings", file=sys.stderr)
+    return 1 if findings else 0
+
+
+if __name__ == "__main__":
+    sys.exit(main())

+"""D-07: SOPS-encrypted secrets files reach checkpoint trees; age private keys and decrypted secrets never do."""
+from __future__ import annotations
+
+import secrets
+import subprocess
+import sys
+from pathlib import Path
+
+import pytest
+
+from scripts.ci import dockerfiles, secret_scan
+
+ROOT = Path(__file__).resolve().parents[2]
+BECH32 = "QPZRY9X8GF2TVDW0S3JN54KHCE6MUA7L"
+ENC = "ENC[AES256_GCM,data:c3ludGhldGlj,iv:AAAAAAAAAAAAAAAAAAAAAA==,tag:AAAAAAAAAAAAAAAAAAAAAA==,type:str]"
+SOPS_META = ("sops:\n  age:\n    - recipient: age1" + "q" * 58 + "\n      enc: |\n"
+             "        -----BEGIN AGE ENCRYPTED FILE-----\n        c3ludGhldGlj\n        -----END AGE ENCRYPTED FILE-----\n"
+             "  lastmodified: '2026-10-19T00:00:00Z'\n  mac: " + ENC + "\n  version: 3.9.0\n")
+
+
+def _synthetic_age_identity() -> str:
+    # random characters in the identity format; no valid checksum, never a usable key
+    return "AGE-SECRET-" + "KEY-1" + "".join(secrets.choice(BECH32) for _ in range(58))
+
+
+def _ignored(path: str) -> bool:
+    r = subprocess.run(["git", "-C", str(ROOT), "check-ignore", "--no-index", "-q", path], capture_output=True, text=True)
+    assert r.returncode in (0, 1), r.stderr
+    return r.returncode == 0
+
+
+@pytest.mark.parametrize("path", ["deploy/secrets/staging.sops.yaml", "deploy/secrets/production.sops.yaml",
+                                  "deploy/secrets/.sops.yaml", "deploy/secrets/README.md",
+                                  "deploy/secrets/escrow-recipients.txt",
+                                  "deploy/secrets/history/staging/20261019T000000Z.sops.yaml"])
+def test_d07_encrypted_secrets_files_are_visible_to_checkpoints(path: str) -> None:
+    assert not _ignored(path), f"{path} is gitignored, so checkpoint trees would drop it"
+
+
+@pytest.mark.parametrize("path", ["deploy/secrets/staging.yaml", "deploy/secrets/staging.dec.yaml",
+                                  "deploy/secrets/keys.txt", "deploy/secrets/staging.agekey",
+                                  "deploy/secrets/history/staging/plain.yaml", "keys.txt",
+                                  "ops/staging.agekey", "secrets/x.txt"])
+def test_d07_age_keys_and_decrypted_files_stay_ignored(path: str) -> None:
+    assert _ignored(path), f"{path} must stay ignored"
+
+
+def test_d07_build_contexts_exclude_sops_files_and_age_keys() -> None:
+    ignore = ROOT / ".dockerignore"
+    for rel in ("deploy/secrets/staging.sops.yaml", "deploy/secrets/staging.yaml", "ops/staging.agekey", "keys.txt"):
+        assert dockerfiles.dockerignored(rel, ignore), rel
+
+
+def test_d07_scan_flags_age_keys_stray_files_and_plaintext(tmp_path: Path) -> None:
+    (tmp_path / "deploy" / "secrets").mkdir(parents=True)
+    (tmp_path / "notes.txt").write_text("key " + _synthetic_age_identity() + "\n", encoding="utf-8")
+    (tmp_path / "deploy/secrets/staging.sops.yaml").write_text("AUTH_SECRET_KEY: SYNTHETIC-PLAINTEXT\n" + SOPS_META, encoding="utf-8")
+    (tmp_path / "deploy/secrets/production.sops.yaml").write_text("AUTH_SECRET_KEY: " + ENC + "\n", encoding="utf-8")
+    (tmp_path / "deploy/secrets/staging.yaml").write_text("AUTH_SECRET_KEY: SYNTHETIC-PLAINTEXT\n", encoding="utf-8")
+    (tmp_path / "ops.agekey").write_text("synthetic\n", encoding="utf-8")
+    paths = ["notes.txt", "deploy/secrets/staging.sops.yaml", "deploy/secrets/production.sops.yaml",
+             "deploy/secrets/staging.yaml", "ops.agekey"]
+    findings = secret_scan.scan(tmp_path, paths)
+    assert {(f.rule, f.path) for f in findings} == {
+        ("age-private-key", "notes.txt"), ("sops-plaintext-value", "deploy/secrets/staging.sops.yaml"),
+        ("sops-metadata-missing", "deploy/secrets/production.sops.yaml"),
+        ("stray-file-in-deploy-secrets", "deploy/secrets/staging.yaml"), ("age-key-file", "ops.agekey")}
+    assert all("SYNTHETIC-PLAINTEXT" not in str(f) and "KEY-1" not in str(f) for f in findings)  # names only, never content
+
