+"""MVP-008: list `tracked file -> untracked dependency` pairs. Stdlib only.
+
+    python scripts/ci/check_tracked_closure.py [--repo DIR] [--report | --strict]
+
+A file that git tracks but that depends on a file git does not track works in this working tree and breaks in any
+clean checkout of the tracked content. Three kinds of dependency are followed, each resolved with only
+`git ls-files` (tracked, the index) and `git ls-files --others --exclude-standard` (untracked, not ignored):
+
+* Python imports of tracked .py files, resolved against the repo root, `src/` and each `services/<svc>`, plus
+  relative imports;
+* the COPY/ADD sources of tracked Dockerfiles (paths relative to the repo root, the build context);
+* `-r` / `-c` includes in tracked requirements files (relative to the including file).
+
+Each pair prints as `<tracked file> -> <untracked file>`. `--report` (default) always exits 0; `--strict` exits 1
+when any pair exists. D-17: nothing is committed, so the real repository shows many pairs for the whole program;
+the list is input for the owner's post-program commit decision and gates nothing.
+"""
+from __future__ import annotations
+
+import argparse
+import ast
+import fnmatch
+import re
+import subprocess
+import sys
+from pathlib import Path, PurePosixPath
+
+if __package__ in (None, ""):  # allow `python scripts\ci\check_tracked_closure.py`
+    sys.path.insert(0, str(Path(__file__).resolve().parents[2]))

+from scripts.ci import dockerfiles  # noqa: E402

+_REQ_INCLUDE = re.compile(r"^\s*(?:-r|-c|--requirement|--constraint)[\s=]+(\S+)")
+_DOCKERFILE = re.compile(r"(^|/)(Dockerfile(\..+)?|[^/]+\.Dockerfile)$")
+_REQUIREMENTS = re.compile(r"(^|/)(requirements[^/]*\.(txt|in)|constraints[^/]*\.txt)$")

+def _ls(repo: Path, *args: str) -> set[str]:
+    proc = subprocess.run(["git", "-C", str(repo), "ls-files", "-z", *args], capture_output=True)
+    if proc.returncode:
+        raise SystemExit(f"git ls-files failed in {repo}: {proc.stderr.decode(errors='replace').strip()}")
+    return {p for p in proc.stdout.decode("utf-8", errors="replace").split("\0") if p}
+
+
+def _norm(path: PurePosixPath) -> str:
+    parts: list[str] = []
+    for part in path.parts:
+        if part == "..":
+            if parts:
+                parts.pop()
+        elif part != ".":
+            parts.append(part)
+    return "/".join(parts)
+
+
+class Closure:
+    def __init__(self, repo: Path) -> None:
+        self.repo = repo
+        self.tracked = _ls(repo)
+        self.untracked = _ls(repo, "--others", "--exclude-standard")
+        self.known = self.tracked | self.untracked
+        services = sorted({p.split("/")[1] for p in self.known if p.startswith("services/") and p.count("/") >= 2})
+        self.roots = ["", "src", *[f"services/{s}" for s in services]]
+
+    def _join(self, root: str, rel: str) -> str:
+        return f"{root}/{rel}" if root else rel
+
+    def _module_files(self, root: str, parts: list[str], leaf_only: bool = False) -> list[str]:
+        """Files that make up `import a.b.c` below `root`: each package __init__ and the module file itself."""
+        out: list[str] = []
+        for i in range(len(parts) if leaf_only else 1, len(parts) + 1):
+            if leaf_only and i != len(parts):
+                continue
+            base = self._join(root, "/".join(parts[:i]))
+            out += [base + ".py", base + "/__init__.py"]
+        return out
+
+    def _resolve(self, parts: list[str], names: list[str]) -> set[str]:
+        for root in self.roots:
+            chain = self._module_files(root, parts) if parts else []
+            if not any(c in self.known for c in chain):
+                continue
+            found = [c for c in chain if c in self.known]
+            for name in names:
+                found += [c for c in self._module_files(root, [*parts, name], leaf_only=True) if c in self.known]
+            return set(found)
+        return set()
+
+    def python_deps(self, path: str) -> set[str]:
+        try:
+            tree = ast.parse((self.repo / path).read_text(encoding="utf-8", errors="replace"))
+        except (SyntaxError, ValueError, OSError):
+            return set()
+        deps: set[str] = set()
+        here = PurePosixPath(path).parent
+        for node in ast.walk(tree):
+            if isinstance(node, ast.Import):
+                for alias in node.names:
+                    deps |= self._resolve(alias.name.split("."), [])
+            elif isinstance(node, ast.ImportFrom):
+                names = [a.name for a in node.names if a.name != "*"]
+                if node.level:
+                    base = here
+                    for _ in range(node.level - 1):
+                        base = base.parent
+                    parts = node.module.split(".") if node.module else []
+                    root = _norm(base)
+                    chain = self._module_files(root, parts) if parts else []
+                    found = [c for c in chain if c in self.known]
+                    if not parts:
+                        found += [self._join(root, "__init__.py")] if self._join(root, "__init__.py") in self.known else []
+                    for name in names:
+                        found += [c for c in self._module_files(root, [*parts, name], leaf_only=True) if c in self.known]
+                    deps |= set(found)
+                else:
+                    deps |= self._resolve(node.module.split(".") if node.module else [], names)
+        return deps
+
+    def dockerfile_deps(self, path: str) -> set[str]:
+        try:
+            df = dockerfiles.parse(self.repo / path)
+        except (OSError, ValueError):
+            return set()
+        deps: set[str] = set()
+        for copy in df.copies:
+            if "from" in copy.flags:
+                continue
+            for src in copy.sources:
+                if "://" in src:
+                    continue
+                rel = _norm(PurePosixPath(src))
+                if rel == "":
+                    deps |= self.untracked
+                    continue
+                for candidate in self.untracked:
+                    if candidate == rel or candidate.startswith(rel.rstrip("/") + "/") or fnmatch.fnmatchcase(candidate, rel):
+                        deps.add(candidate)
+        return deps
+
+    def requirements_deps(self, path: str) -> set[str]:
+        deps: set[str] = set()
+        for line in (self.repo / path).read_text(encoding="utf-8", errors="replace").splitlines():
+            match = _REQ_INCLUDE.match(line)
+            if match:
+                deps.add(_norm(PurePosixPath(path).parent / match.group(1)))
+        return deps & self.untracked
+
+    def pairs(self) -> list[tuple[str, str]]:
+        found: set[tuple[str, str]] = set()
+        for path in sorted(self.tracked):
+            if not (self.repo / path).is_file():
+                continue
+            deps: set[str] = set()
+            if path.endswith(".py"):
+                deps |= self.python_deps(path)
+            if _DOCKERFILE.search(path):
+                deps |= self.dockerfile_deps(path)
+            if _REQUIREMENTS.search(path):
+                deps |= self.requirements_deps(path)
+            found |= {(path, dep) for dep in deps if dep in self.untracked and dep != path}
+        return sorted(found)
+
+
+def main(argv: list[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
+    parser.add_argument("--repo", default=str(Path(__file__).resolve().parents[2]))
+    mode = parser.add_mutually_exclusive_group()
+    mode.add_argument("--report", action="store_true", help="print the pairs, always exit 0 (default)")
+    mode.add_argument("--strict", action="store_true", help="exit 1 when any pair exists")
+    args = parser.parse_args(argv)
+    pairs = Closure(Path(args.repo)).pairs()
+    for tracked, untracked in pairs:
+        print(f"{tracked} -> {untracked}")
+    print(f"{len(pairs)} tracked -> untracked pair(s)")
+    return 1 if (args.strict and pairs) else 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+"""MVP-008: build and probe every registered image from a clean export of a git tree. Stdlib only.
+
+    python scripts/ci/clean_checkout.py --tree <tree id> [--out DIR] [--compose] [--tag TAG]
+                                        [--tests] [--remove-images]
+
+D-17: nothing is committed, so the checkout is ``git archive --format=tar <tree>`` of a checkpoint tree (read-only
+for the repository), extracted into ``D:\\tmp\\mvp-exec\\clean-checkout\\<tree7>`` (or ``--out``). Every ``IMAGES``
+entry is built from that directory with ``build_images.build_command`` and probed by importing its modules inside
+the image. Exit 0 only when every step succeeds; a failure never stops the later steps, so one run lists every
+broken image.
+
+The export has no ``.git``. The tests that call ``git ls-files`` or ``git check-ignore`` (tests/build) would exit 2
+there, so ``--tests`` first gives the export a private git context (``git init`` plus ``git add -A`` inside the
+export directory, never a commit and never touching this repository) and then runs tests/build in it.
+"""
+from __future__ import annotations
+
+import argparse
+import io
+import os
+import shutil
+import subprocess
+import sys
+import tarfile
+from collections.abc import Sequence
+from dataclasses import dataclass, field
+from pathlib import Path
+
+if __package__ in (None, ""):  # allow `python scripts\ci\clean_checkout.py`
+    sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
+
+from scripts.ci.build_images import build_command, image_ref  # noqa: E402
+from scripts.ci.images import IMAGES, Image, probe_env  # noqa: E402
+
+REPO = Path(__file__).resolve().parents[2]
+DEFAULT_ROOT = Path(r"D:\tmp\mvp-exec\clean-checkout")
+MARKER = ".clean-checkout-export"
+TAIL_LINES = 25
+
+
+@dataclass(frozen=True)
+class Step:
+    image: Image
+    build: list[str]
+    probes: list[list[str]] = field(default_factory=list)
+
+
+def probe_commands(image: Image, tag: str, env: dict[str, str] | None = None) -> list[list[str]]:
+    """One `docker run` per module: the CMD target (if any) first, then the image's probe modules."""
+    env_flags: list[str] = []
+    for key, value in (env if env is not None else probe_env()).items():
+        env_flags += ["-e", f"{key}={value}"]
+    modules = ([image.module] if image.module else []) + list(image.probes)
+    return [
+        ["docker", "run", "--rm", "--entrypoint", "python", "-e", "ENVIRONMENT=development", *env_flags,
+         image_ref(image.name, tag), "-c", f"import importlib; importlib.import_module({module!r})"]
+        for module in modules
+    ]
+
+
+def plan(tag: str, images: Sequence[Image] = IMAGES) -> list[Step]:
+    env = probe_env()  # one throwaway set per run
+    return [Step(i, build_command(i, tag, "."), probe_commands(i, tag, env)) for i in images]
+
+
+def has_git_context(path: Path) -> bool:
+    """True when `path` is the top of a git work tree; a `git archive` export is not."""
+    return (Path(path) / ".git").exists()
+
+
+def _printable(cmd: Sequence[str]) -> str:
+    """The command for the log, with the values of probe `-e KEY=value` flags hidden."""
+    shown: list[str] = []
+    hide = False
+    for part in cmd:
+        if hide:
+            key = part.partition("=")[0]
+            shown.append(part if key == "ENVIRONMENT" else f"{key}=***")
+            hide = False
+            continue
+        hide = part == "-e"
+        shown.append(part)
+    return " ".join(shown)
+
+
+def export_tree(tree: str, out: Path, repo: Path = REPO) -> None:
+    """Extract `git archive --format=tar <tree>` into an empty (or previous export) directory."""
+    if out.exists():
+        if any(out.iterdir()) and not (out / MARKER).exists():
+            raise SystemExit(f"refusing to reuse {out}: it is not empty and is not an earlier export")
+        shutil.rmtree(out)
+    out.mkdir(parents=True)
+    proc = subprocess.run(["git", "-C", str(repo), "archive", "--format=tar", tree], capture_output=True)
+    if proc.returncode:
+        raise SystemExit(f"git archive {tree} failed: {proc.stderr.decode(errors='replace').strip()}")
+    with tarfile.open(fileobj=io.BytesIO(proc.stdout)) as tar:
+        tar.extractall(out, filter="data")
+    (out / MARKER).write_text(tree + "\n", encoding="utf-8")
+
+
+def _run(label: str, cmd: list[str], cwd: Path, results: list[tuple[str, bool]]) -> bool:
+    proc = subprocess.run(cmd, cwd=str(cwd), capture_output=True, text=True, errors="replace")
+    ok = proc.returncode == 0
+    print(f"[{'ok' if ok else 'FAIL'}] {label}", flush=True)
+    if not ok:
+        print(f"      $ {_printable(cmd)}")
+        tail = (proc.stdout + proc.stderr).strip().splitlines()[-TAIL_LINES:]
+        for line in tail:
+            print(f"      | {line}")
+    results.append((label, ok))
+    return ok
+
+
+def _git_context(out: Path) -> None:
+    """A private repository inside the export so `git ls-files` works; nothing is committed."""
+    for args in (["init", "-q"], ["add", "-A"]):
+        subprocess.run(["git", "-c", "core.safecrlf=false", "-C", str(out), *args], check=True, capture_output=True)
+
+
+def run(tree: str, out: Path, tag: str, compose: bool, tests: bool, remove_images: bool) -> int:

+    print(f"exported {tree} -> {out}", flush=True)
+    results: list[tuple[str, bool]] = []
+    steps = plan(tag)
+    try:
+        for step in steps:
+            built = _run(f"build {step.image.name}", step.build, out, results)
+            for probe in step.probes if built else []:
+                _run(f"probe {step.image.name}: {probe[-1].split('import_module(')[-1].rstrip(')')}", probe, out, results)
+        if compose:
+            _run("compose build", ["docker", "compose", "-f", "docker-compose.yml", "build"], out, results)
+        if tests:

+            env_cmd = [sys.executable, "-m", "pytest", "tests/build", "-o", "addopts=", "-p", "no:logging", "-q"]
